        console.error(
          "MA7ALAK Stories:",
          result.error
        );

        return;

      }


      stories =
        result.data || [];

      storiesLoaded =
        true;

      /*
         Do not download Story media here.
         The permanent circle image is enough until the visitor opens
         Stories. Keep only active warm-cache entries and schedule the
         exact next 24-hour expiry refresh.
      */
      pruneStoryWarmCache();
      scheduleStoryExpiryRefresh();


      if(storyWrapper){

        storyWrapper.classList.toggle(
          "story-has-story",
          stories.length > 0
        );

      }


      /*
         Keep the same permanent logo whether there are
         zero Stories, image Stories, or video Stories.
      */
      renderStoryCircleImage();


      /*
         The existing new-Story gold animation still works.
      */
      updateNewStoryAnimation();

    }

    catch(error){

      console.error(
        "MA7ALAK Stories:",
        error
      );

    }

  }

  function requestStoryDeleteFromParent(story){

    return new Promise(function(resolve,reject){

      if(
        !window.parent ||
        window.parent === window
      ){
        reject(
          new Error(
            "STORY_DELETE_PARENT_UNAVAILABLE"
          )
        );
        return;
      }

      const requestId =
        "story-delete-" +
        Date.now() +
        "-" +
        Math.random()
          .toString(36)
          .slice(2);

      let timeout = null;

      function cleanup(){
        window.removeEventListener(
          "message",
          onMessage
        );

        if(timeout){
          clearTimeout(timeout);
          timeout = null;
        }
      }

      function onMessage(event){

        if(
          event.source !== window.parent ||
          !event.data ||
          event.data.type !==
            "MA7ALAK_DELETE_STORY_RESULT" ||
          String(event.data.requestId || "") !==
            requestId
        ){
          return;
        }

        cleanup();

        if(event.data.ok){
          resolve(true);
        }
        else{
          reject(
            new Error(
              event.data.error ||
              "Could not delete Story."
            )
          );
        }

      }

      window.addEventListener(
        "message",
        onMessage
      );

      timeout =
        setTimeout(
          function(){
            cleanup();
            reject(
              new Error(
                "Story deletion timed out. Please try again."
              )
            );
          },
          15000
        );

      try{
        window.parent.postMessage(
          {
            type:
              "MA7ALAK_DELETE_STORY_REQUEST",
            requestId:
              requestId,
            shopSlug:
              SHOP_SLUG,
            storyId:
              story.id
          },
          "*"
        );
      }
      catch(error){
        cleanup();
        reject(error);
      }

    });

  }


  async function deleteStoryDirect(story){

    const client =
      await getMa7alakSupabaseClient();

    const userResult =
      await client.auth.getUser();

    const user =
      userResult &&
      userResult.data &&
      userResult.data.user;

    if(!user){
      throw new Error(
        "You need to be signed in."
      );
    }

    const ownerResult =
      await client
        .from("shop_owners")
        .select("shop_slug")
        .eq("user_id", user.id)
        .eq("shop_slug", SHOP_SLUG)
        .maybeSingle();

    if(
      ownerResult.error ||
      !ownerResult.data
    ){
      throw ownerResult.error ||
        new Error(
          "You no longer have permission to manage this shop."
        );
    }

    const storyResult =
      await client
        .from("shop_stories")
        .select("id,storage_path")
        .eq("id", story.id)
        .eq("shop_slug", SHOP_SLUG)
        .maybeSingle();

    if(storyResult.error){
      throw storyResult.error;
    }

    if(!storyResult.data){
      return;
    }

    const databaseResult =
      await client
        .from("shop_stories")
        .delete()
        .eq("id", story.id)
        .eq("shop_slug", SHOP_SLUG);

    if(databaseResult.error){
      throw databaseResult.error;
    }

    const storagePath =
      String(
        storyResult.data.storage_path ||
        ""
      ).trim();

    if(storagePath){
      const storageResult =
        await client
          .storage
          .from("shop-stories")
          .remove([storagePath]);

      if(storageResult.error){
        console.warn(
          "MA7ALAK Story storage cleanup:",
          storageResult.error
        );
      }
    }

  }


  /* =========================================================
     DELETE CURRENT STORY
  ========================================================= */

  async function deleteCurrentStory(){

    if(
      !isShopOwner ||
      isDeletingStory ||
      !stories.length
    ){
      return;
    }

    const story =
      stories[
        currentStory
      ];

    if(!story){
      return;
    }

    /*
       Start the delete-flow guard BEFORE confirm(). Safari / Samsung
       browsers may leave native fullscreen while the confirm dialog is
       open and dispatch fullscreenchange after the dialog returns.
    */
    storyDeleteFlowActive =
      true;

    const confirmed =
      window.confirm(
        "Delete this Story?\n\nThis cannot be undone."
      );

    if(!confirmed){

      /*
         Keep the CSS viewer open even if native fullscreen was dropped
         by the browser while the confirmation dialog was visible.
      */
      const screen =
        document.getElementById(
          "ma7alak-full-story"
        );

      if(screen){
        screen.classList.add(
          "active"
        );
      }

      document.documentElement.style.overflow =
        "hidden";

      document.body.style.overflow =
        "hidden";

      setTimeout(
        function(){
          storyDeleteFlowActive = false;
        },
        450
      );

      return;
    }

    try{

      isDeletingStory =
        true;

      const deleteButton =
        document.getElementById(
          "ma7alak-story-delete-btn"
        );

      if(deleteButton){
        deleteButton.disabled =
          true;

        deleteButton.textContent =
          "⏳";
      }

      clearInterval(
        timer
      );

      if(
        currentMedia &&
        currentMedia.tagName ===
        "VIDEO"
      ){
        try{
          currentMedia.pause();
          currentMedia.muted =
            true;
        }
        catch(error){}
      }

      /*
         Hostinger Story embeds delete through the top-level authenticated
         page bridge. A top-level Story implementation can still delete
         directly with the shared authenticated Supabase client.
      */
      if(
        window.parent &&
        window.parent !== window
      ){
        await requestStoryDeleteFromParent(
          story
        );
      }
      else{
        await deleteStoryDirect(
          story
        );
      }

      /*
         Realtime may have already refreshed the Story list while the
         delete request was in flight. Remove by Story ID, never blindly by
         current index, so another Story can never be removed accidentally.
      */
      const deletedIndex =
        stories.findIndex(
          function(item){
            return String(item && item.id) ===
              String(story.id);
          }
        );

      if(deletedIndex >= 0){
        stories.splice(
          deletedIndex,
          1
        );

        if(
          deletedIndex <
          currentStory
        ){
          currentStory -= 1;
        }
      }

      pruneStoryWarmCache();
      scheduleStoryExpiryRefresh();

      if(storyWrapper){
        storyWrapper.classList.toggle(
          "story-has-story",
          stories.length > 0
        );
      }

      updateNewStoryAnimation();

      /*
         Stay inside the Story viewer while there are more Stories so the
         owner can continue deleting them one after another.
      */
      if(stories.length > 0){

        if(
          currentStory >=
          stories.length
        ){
          currentStory =
            stories.length - 1;
        }

        if(currentStory < 0){
          currentStory = 0;
        }

        /*
           Native fullscreen may have been dropped by the confirmation
           dialog. The Story viewer itself is fixed-position fullscreen,
           so explicitly keep it active and page-locked instead of
           closing back to the shop page.
        */
        const screen =
          createStoryScreen();

        screen.style.position =
          "fixed";

        screen.style.left =
          "0";

        screen.style.top =
          "0";

        screen.style.width =
          "100vw";

        screen.style.height =
          "100vh";

        screen.classList.add(
          "active"
        );

        document.documentElement.style.overflow =
          "hidden";

        document.body.style.overflow =
          "hidden";

        createProgressBars();

        showStory(
          currentStory
        );

        return;
      }

      /* No Stories remain, so there is nothing else to manage. */
      updateDeleteButton();
      closeStories();

    }

    catch(error){

      console.error(
        "MA7ALAK delete story error:",
        error
      );

      alert(
        error && error.message
          ? error.message
          : "Could not delete Story. Please try again."
      );

    }

    finally{

      isDeletingStory =
        false;

      /*
         Leave a short guard window for queued fullscreenchange events
         generated by the browser's confirmation dialog.
      */
      setTimeout(
        function(){
          storyDeleteFlowActive = false;
        },
        450
      );

      const deleteButton =
        document.getElementById(
          "ma7alak-story-delete-btn"
        );

      if(deleteButton){
        deleteButton.disabled =
          false;

        deleteButton.textContent =
          "🗑️";
      }

      updateDeleteButton();

    }

  }

  function restoreStoryFrameViewport(){

    const restore = ma7alakStoryFrameRestore;
    ma7alakStoryFrameRestore = null;

    if(!restore){
      return;
    }

    function restoreStyle(node,value){
      if(!node)return;
      if(value === null)node.removeAttribute("style");
      else node.setAttribute("style",value);
    }

    try{
      restoreStyle(restore.frame,restore.frameStyle);
      restoreStyle(restore.host,restore.hostStyle);
      (restore.ancestors||[]).forEach(function(item){
        restoreStyle(item.node,item.style);
      });
      restoreStyle(restore.frame.ownerDocument.documentElement,restore.htmlStyle);
      restoreStyle(restore.frame.ownerDocument.body,restore.bodyStyle);
    }
    catch(error){}

  }


  /* =========================================================
     OPEN STORIES
  ========================================================= */

  async function openStories(){

    if(ma7alakStoryOpenPromise){
      return ma7alakStoryOpenPromise;
    }

    /*
       If the Story list already finished loading and it is empty,
       do not activate the black fullscreen viewer at all.
    */
    if(
      storiesLoaded &&
      stories.length === 0
    ){
      alert("No Stories right now 📸");
      return false;
    }

    /*
       REAL fullscreen first.

       On Android/Chrome/Brave this is the only browser-supported way to
       hide the address/tab bars. The Hostinger viewport portal remains only
       as a fallback when element fullscreen is unsupported or rejected.
    */
    const screen =
      createStoryScreen();

    screen.style.position = "fixed";
    screen.style.left = "0";
    screen.style.top = "0";
    screen.style.width = "100vw";
    screen.style.height = "100vh";
    screen.style.height = "100dvh";
    screen.classList.add("active");

    document.documentElement.style.overflow = "hidden";
    document.body.style.overflow = "hidden";

    let nativeFullscreenRequested =
      false;

    try{

      const request =
        screen.requestFullscreen ||
        screen.webkitRequestFullscreen;


      if(
        typeof request ===
        "function"
      ){

        nativeFullscreenRequested =
          true;


        const result =
          request.call(
            screen,
            {
              navigationUI:
                "hide"
            }
          );


        if(
          result &&
          typeof result.catch ===
            "function"
        ){

          result.catch(
            function(){

              try{
                if(window.parent && window.parent !== window){
                  window.parent.postMessage(
                    {type:"SHOUFHON_EMBED_VIEWER_OPEN"},
                    "*"
                  );
                }
              }
              catch(error){}

            }
          );

        }

      }

    }
    catch(error){

      nativeFullscreenRequested =
        false;

    }


    if(
      !nativeFullscreenRequested
    ){

      try{
        if(window.parent && window.parent !== window){
          window.parent.postMessage(
            {type:"SHOUFHON_EMBED_VIEWER_OPEN"},
            "*"
          );
        }
      }
      catch(error){}

    }

    ma7alakStoryOpenPromise = (async function(){

      if(stories.length === 0){
        const status=document.getElementById("ma7alak-story-status-text");
        if(status)status.textContent="Loading Story…";

        try{
          await loadStories();
        }
        catch(error){}
      }

      if(stories.length === 0){
        closeStories();
        setTimeout(function(){
          alert("No Stories right now 📸");
        },80);
        return false;
      }

      currentStory =
        getFirstUnseenStoryIndex();

      createProgressBars();
      updateDeleteButton();
      showStory(currentStory);

      return true;

    })();

    try{
      return await ma7alakStoryOpenPromise;
    }
    finally{
      ma7alakStoryOpenPromise = null;
    }

  }

        false;


      media.muted =

      media.playsInline =
        true;

        "auto";


      media.setAttribute(
        "playsinline",
        ""
      );


      media.setAttribute(
        "webkit-playsinline",
        ""
      );


      media.setAttribute(
        "disablepictureinpicture",
        ""
      );


      media.setAttribute(
        "controlslist",
        "nodownload noplaybackrate"
      );

    }

    else{

  /* =========================================================
     STORY VIEW TRACKING
  ========================================================= */

  async function recordCurrentStoryView(){

    const story =
      stories[currentStory];

    if(
      !story ||
      !story.id ||
      !visitorId
    ){
      return;
    }

    const storyId =
      String(story.id);

    if(
      recordedStoryViews.has(
        storyId
      )
    ){
      return;
    }

    /*
       Mark locally first so fast swipes/back-and-forth do not
       fire duplicate requests. The database unique constraint
       is the final protection against duplicate views.
    */
    recordedStoryViews.add(
      storyId
    );

    try{

      const result =
        await supabaseClient.rpc(
          "record_story_view",
          {
            p_story_id:
              story.id,

            p_shop_slug:
              story.shop_slug || SHOP_SLUG,

            p_visitor_id:
              visitorId
          }
        );

      if(result.error){
        throw result.error;
      }

    }
    catch(error){

      /*
         Allow a retry later if the network request failed.
      */
      recordedStoryViews.delete(
        storyId
      );

      console.error(
        "MA7ALAK story view:",
        error
      );

    }

  }

    try{
      await getMa7alakSupabaseClient();

    }


    const box =
      document.getElementById(
        "ma7alak-story-view-count"
      );

    if(!box){
      return;
    }

    /*
       Only the verified shop owner can ever see this control.
    */
    if(
      !isShopOwner ||
      !stories.length
    ){
      box.classList.remove(
        "visible"
      );
      return;
    }

    const story =
      stories[currentStory];

    if(
      !story ||
      !story.id
    ){
      box.classList.remove(
        "visible"
      );
      return;
    }

    box.classList.add(
      "visible"
    );

    const number =
      box.querySelector(
        ".ma7alak-view-number"
      );

    const label =
      box.querySelector(
        ".ma7alak-view-label"
      );

    try{

      const result =
        await supabaseClient.rpc(
          "get_story_view_count",
          {
            p_story_id:
              story.id,

            p_shop_slug:
              story.shop_slug || SHOP_SLUG
          }
        );

      if(result.error){
        throw result.error;
      }

      const count =
        Math.max(
          0,
          Number(result.data) || 0
        );

        number.textContent =
          String(count);
      }

      if(label){
        label.textContent =
          count === 1
            ? "view"
            : "views";
      }

    }
    catch(error){

      /*
         If Supabase rejects the owner-only count request,
         hide it instead of exposing an incorrect number.
      */
      box.classList.remove(
        "visible"
      );

      console.error(
        "MA7ALAK story view count:",
        error
      );

    }

  }


  /* =========================================================
     STORY LIKE BUTTON STATE
  ========================================================= */

  function updateLikeButton(){

    const button =
      document.getElementById(
        "ma7alak-story-like-btn"
      );

    if(!button){
      return;
    }

    /*
       The top-level owner bridge is authoritative. A shop owner should
       never be offered a Like control on their own Story, even when this
       Hostinger iframe itself has no authenticated Supabase session.
    */
    if(isShopOwner){
      button.style.setProperty("display","none","important");
      button.disabled = true;
      button.dataset.ownerLocked = "1";
      button.setAttribute("aria-hidden","true");
      return;
    }

    if(button.dataset.ownerLocked === "1"){
      button.disabled = false;
      delete button.dataset.ownerLocked;
    }

    button.removeAttribute("aria-hidden");

    const story =
      stories[currentStory];

    if(!story){
      button.style.setProperty("display","none","important");
      return;
    }

    button.style.setProperty("display","flex","important");

    const liked =
      likedStoryIds.has(
        String(story.id)
      );

    const heart =
      button.querySelector(
        ".ma7alak-like-heart"
      );

    if(liked){

      button.classList.add("liked");

      if(heart){
        heart.textContent = "♥";
      }

      button.setAttribute(
        "aria-label",
        "Story liked"
      );

    }
    else{

      button.classList.remove("liked");

      if(heart){
        heart.textContent = "♡";
      }

      button.setAttribute(
        "aria-label",
        "Like story"
      );

    }

  }

  /* =========================================================
     LIKE CURRENT STORY
  ========================================================= */

  async function likeCurrentStory(){

    if(isShopOwner){
      return;
    }

    const story =
      stories[currentStory];

    if(!story || !story.id || !visitorId){
      console.warn(
        "MA7ALAK: cannot like story - missing story ID or visitor ID"
      );
      return;
    }

    const storyId =
      String(story.id);

    const button =
      document.getElementById(
        "ma7alak-story-like-btn"
      );

    /*
      Each story/video has its own ID.
      One visitor can therefore like Story 1, Story 2,
      Story 3, etc. independently.
    */
    if(likedStoryIds.has(storyId)){
      updateLikeButton();
      return;
    }

    if(button){
      button.disabled = true;
      button.style.pointerEvents = "none";
    }

    try{

      const result =
        await supabaseClient.rpc(
          "like_story",
          {
            p_story_id: story.id,
            p_visitor_id: visitorId
          }
        );

      if(result.error){
        throw result.error;
      }

      const data =
        result.data;

      console.log(
        "MA7ALAK story like result:",
        data
      );

      if(!data || data.success !== true){
        throw new Error(
          (data && data.message) ||
          "Like was not accepted by Supabase"
        );
      }

      /*
        Supabase is the source of truth.
        If this visitor already liked this exact story,
        the RPC returns already_liked=true and we still
        show the correct liked state.
      */
      likedStoryIds.add(storyId);
      saveLikedStories();

      updateLikeButton();
      launchLikeHearts();

    }
    catch(error){

      console.error(
        "MA7ALAK story like error:",
        error
      );

      alert(
        "ما قدرنا نسجّل اللايك. جرّب مرة ثانية."
      );

      /*
        Do not permanently mark the story as liked when
        Supabase rejects the request.
      */
      likedStoryIds.delete(storyId);
      saveLikedStories();

    }
    finally{

      if(button){
        button.disabled = false;
        button.style.pointerEvents = "auto";
      }

    }

  }


  /* =========================================================
     SHOW STORY
  ========================================================= */

  function showStory(
    index
  ){

    if(
      !stories.length
    ){

      return;

    }


    clearInterval(
      timer
    );


    if(
      index < 0
    ){

      index =
        stories.length - 1;

    }


    if(
      index >=
      stories.length
    ){

      index =
        0;

    }


    currentStory =
      index;


    markCurrentStoryAsSeen();


    const story =
      stories[
        currentStory
      ];


    const screen =
      document.getElementById(
        "ma7alak-full-story"
      );


    const container =
      document.getElementById(
        "ma7alak-full-story-media"
      );


    if(
      !screen ||
      !container
    ){

      return;

    }


    /*
      Update timestamp immediately.
    */

    updateStoryTime();


    /*
      Make sure owner controls match
      current authentication state.
    */

    updateDeleteButton();
    updateLikeButton();

    /*
       Record one unique view for this exact Story.
       The owner-only eye/count is refreshed separately.
    */
    recordCurrentStoryView();
    updateStoryViewCount();


    const thisTransition =
      ++transitionToken;


    updateProgressBars(
      currentStory,
      0
    );


    if(
      currentMedia &&
      currentMedia.tagName ===
      "VIDEO"
    ){

      try{

        currentMedia.pause();

        currentMedia.muted =
          true;

      }

      catch(error){}

    }


    const result =
      createMediaElement(
        story
      );


    const media =
      result.media;


    const type =
      result.type;


    container.appendChild(
      media
    );


    function activateMedia(){
      if(
        thisTransition !==
        transitionToken
      ){

        if(
          media.tagName ===
          "VIDEO"
        ){

          try{

            media.pause();

            media.muted =
              true;

          }

          catch(error){}

        }


        if(
          media.parentNode
        ){

          media.parentNode.removeChild(
            media
          );

        }


        return;

      }


      const allVideos =
        container.querySelectorAll(
          "video"
        );


      allVideos.forEach(
        function(video){

          if(
            video !==
            media
          ){

            try{

              video.pause();

              video.muted =
                true;

            }

            catch(error){}

          }

        }
      );


      media.style.visibility =
        "visible";


      if(
        type === "video"
      ){

        media.muted =
          false;


        const playPromise =
          media.play();


        if(
          playPromise &&
          typeof playPromise.catch ===
          "function"
        ){

          playPromise.catch(
            function(){

              const activeScreen =
                document.getElementById(
                  "ma7alak-full-story"
                );

              if(
                thisTransition !== transitionToken ||
                !activeScreen ||
                !activeScreen.classList.contains("active") ||
                !media.isConnected
              ){
                try{
                  media.pause();
                  media.muted = true;
                  media.volume = 0;
                }
                catch(error){}
                return;
              }

              media.muted = true;
              media.play().catch(function(){});

            }
          );

        }

      }


      requestAnimationFrame(
        function(){

          if(
            thisTransition !==
            transitionToken
          ){

            return;

          }


          media.style.opacity =
            "1";


          media.classList.add(
            "active"
          );


          const oldMedia =
            currentMedia;


          currentMedia =
            media;


          /*
             The visitor is actively watching Stories now, so warming
             only the next Story is useful instead of preloading the
             whole Story set on page load.
          */
          warmNextStoryMedia();


          if(
            oldMedia &&
            oldMedia !==
            media
          ){

            oldMedia.style.opacity =
              "0";


            oldMedia.classList.remove(
              "active"
            );


            setTimeout(
              function(){

                if(
                  oldMedia &&
                  oldMedia !==
                  currentMedia
                ){

                  if(
                    oldMedia.tagName ===
                    "VIDEO"
                  ){

                    try{

                      oldMedia.pause();

                      oldMedia.muted =
                        true;

                      oldMedia.removeAttribute(
                        "src"
                      );

                      oldMedia.load();

                    }

                    catch(error){}

                  }


                  if(
                    oldMedia.parentNode
                  ){

                    oldMedia.parentNode.removeChild(
                      oldMedia
                    );

                  }

                }

              },
              360
            );

          }

        }
      );

    }


    if(
      type === "video"
    ){

      let activated =
        false;


      function activateVideoOnce(){

        if(activated){

          return;

        }


        activated =
          true;


        activateMedia();

      }


      media.addEventListener(
        "loadedmetadata",
        function(){
          /*
             Asking for the first frame as soon as metadata exists
             helps mobile browsers render fresh uploads sooner.
          */
          try{
            if(
              media.readyState < 2 &&
              media.currentTime === 0
            ){
              media.currentTime =
                0.001;
            }
          }
          catch(error){}
        },
        {
          once:true
        }
      );


      media.addEventListener(
        "loadeddata",
        activateVideoOnce,
        {
          once:true
        }
      );


      media.addEventListener(
        "canplay",
        activateVideoOnce,
        {
          once:true
        }
      );


      if(
        media.readyState >=
        2
      ){

        activateVideoOnce();

      }


      media.addEventListener(
        "timeupdate",
        function(){

          if(
            media.duration &&
            isFinite(
              media.duration
            )
          ){

            updateProgressBars(
              currentStory,
              (
                media.currentTime /
                media.duration
              ) *
              100
            );

          }

        }
      );


      media.addEventListener(
        "ended",
        function(){

          if(
            thisTransition ===
            transitionToken &&
            currentMedia ===
            media
          ){

            nextStory();

          }

        }
      );

    }


    else{

      let activated =
        false;


      function activateImageOnce(){

        if(activated){

          return;

        }


        activated =
          true;


        activateMedia();


        const started =
          Date.now();


        timer =
          setInterval(
            function(){

              if(
                thisTransition !==
                transitionToken
              ){

                clearInterval(
                  timer
                );

                return;

              }


              const elapsed =
                Date.now() -
                started;


              const percent =
                Math.min(
                  100,
                  (
                    elapsed /
                    5000
                  ) *
                  100
                );


              updateProgressBars(
                currentStory,
                percent
              );


              if(
                percent >=
                100
              ){

                clearInterval(
                  timer
                );


                nextStory();

              }

            },
            50
          );

      }


      media.addEventListener(
        "load",
        activateImageOnce,
        {
          once:true
        }
      );


      if(
        media.complete
      ){

        activateImageOnce();

      }

    }

  }


  /* =========================================================
     NEXT
  ========================================================= */

  function nextStory(){

    if(
      !stories.length ||
      isDeletingStory
    ){

      return;

    }


    if(
      currentStory >=
      stories.length - 1
    ){

      closeStories();
      return;

    }


    currentStory++;


    showStory(
      currentStory
    );

  }


  /* =========================================================
     PREVIOUS
  ========================================================= */

  function previousStory(){

    if(
      !stories.length ||
      isDeletingStory
    ){

      return;

    }


    if(currentStory <= 0){
      currentStory = 0;
      showStory(0);
      return;
    }


    currentStory--;


    showStory(
      currentStory
    );

  }


  function stopStoryPlayback(unloadMedia){

    clearInterval(timer);
    transitionToken++;

    if(currentMedia && currentMedia.tagName === "VIDEO"){
      try{
        currentMedia.pause();
        currentMedia.muted = true;
        currentMedia.volume = 0;
        if(unloadMedia){
          currentMedia.removeAttribute("src");
          currentMedia.load();
        }
      }
      catch(error){}
    }

    const screen = document.getElementById("ma7alak-full-story");
    if(screen){
      screen.querySelectorAll("video").forEach(function(video){
        try{
          video.pause();
          video.muted = true;
          video.volume = 0;
          if(unloadMedia){
            video.removeAttribute("src");
            video.load();
          }
        }
        catch(error){}
      });
    }

    currentMedia = null;
  }


  /* =========================================================
     CLOSE STORIES
  ========================================================= */

  function closeStories(){

    const screen =
      document.getElementById(
        "ma7alak-full-story"
      );

    if(!screen || ma7alakStoryClosing){
      return;
    }

    ma7alakStoryClosing = true;
    stopStoryPlayback(true);

    try{

      const fs =
        document.fullscreenElement ||
        document.webkitFullscreenElement;


      if(
        fs === screen ||
        (
          fs &&
          screen.contains(fs)
        )
      ){

        const exit =
          document.exitFullscreen ||
          document.webkitExitFullscreen;


        if(
          typeof exit ===
          "function"
        ){

          const result =
            exit.call(
              document
            );


          if(
            result &&
            typeof result.catch ===
              "function"
          ){
            result.catch(
              function(){}
            );
          }

        }

      }

    }
    catch(error){}

    screen.classList.remove("active");

    /* Always unlock this embed immediately. */
    document.documentElement.style.removeProperty("overflow");
    document.body.style.removeProperty("overflow");
    document.documentElement.style.removeProperty("touch-action");
    document.body.style.removeProperty("touch-action");
    document.documentElement.style.removeProperty("overscroll-behavior");
    document.body.style.removeProperty("overscroll-behavior");

    /*
       Parent portal owns the parent-page lock. Tell it to release BEFORE any
       local cleanup so phone scrolling is restored even if the iframe changes.
    */
    try{
      if(window.parent && window.parent !== window){
        window.parent.postMessage(
          {type:"SHOUFHON_EMBED_VIEWER_CLOSE"},
          "*"
        );
      }
    }
    catch(error){}

    /* Backward compatibility only; V5 no longer performs direct promotion. */
    restoreStoryFrameViewport();

    setTimeout(function(){
      if(!screen.classList.contains("active")){
        const container =
          document.getElementById("ma7alak-full-story-media");

        if(container){
          container.innerHTML = "";
        }

        currentMedia = null;
      }

      ma7alakStoryClosing = false;
    },80);
  }

function ma7alakResolveShopSlug(){

  const explicit =
    document
      .getElementById("ma7alak-profile-shell")
      ?.getAttribute("data-shop-slug") ||
    "";

  if(String(explicit).trim()){
    return String(explicit).trim().toLowerCase();
  }

  const candidates = [];

  try{
    if(window.parent && window.parent !== window){
      candidates.push(window.parent.location.pathname || "");
    }
  }catch(_){}

  try{
    if(document.referrer){
      candidates.push(new URL(document.referrer).pathname || "");
    }
  }catch(_){}

  candidates.push(window.location.pathname || "");

  for(const pathname of candidates){
    const parts =
      String(pathname)
        .split("/")
        .filter(Boolean);

    const last =
      parts.length
        ? decodeURIComponent(parts[parts.length - 1])
        : "";

    if(last && last !== "embed" && last !== "editor"){
      return last.trim().toLowerCase();
    }
  }

  return "";
}

/* =========================================================
   VISITOR ID
========================================================= */

function randomPart(){

  if(
    window.crypto
      ?.getRandomValues
  ){

    const bytes =
      new Uint8Array(16);


    window.crypto
      .getRandomValues(
        bytes
      );


    return Array
      .from(
        bytes,
        function(value){

          return value
            .toString(16)
            .padStart(
              2,
              "0"
            );

        }
      )
      .join("");

  }


  return (

    Math.random()
      .toString(36)
      .slice(2)

    +

    Math.random()
      .toString(36)
      .slice(2)

  );

}

/* =========================================================
   GET VISITOR ID
========================================================= */

function getVisitorId(){

  const existing =
    readStorage();


  if(
    existing &&
    existing.length >= 8 &&
    existing.length <= 200
  ){

    return existing;

  }


  const value =

    "v_"

    +

    Date.now()
      .toString(36)

    +

    "_"

    +

    randomPart();


  writeStorage(
    value
  );


  return value;

}

function m7HexToRgb(hex){

  const clean =
    String(hex || "")
      .trim()
      .replace("#","");

  if(!/^[0-9a-f]{6}$/i.test(clean)){
    return null;
  }

  return {
    r:parseInt(clean.slice(0,2),16),
    g:parseInt(clean.slice(2,4),16),
    b:parseInt(clean.slice(4,6),16)
  };

}


function m7Mix(rgb,target,amount){

  const clamp =
    function(number){
      return Math.max(
        0,
        Math.min(
          255,
          Math.round(number)
        )
      );
    };

  return "#" +
    [
      clamp(rgb.r + (target.r-rgb.r)*amount),
      clamp(rgb.g + (target.g-rgb.g)*amount),
      clamp(rgb.b + (target.b-rgb.b)*amount)
    ]
      .map(function(value){
        return value
          .toString(16)
          .padStart(2,"0");
      })
      .join("");

}


function m7ApplyAccent(hex){

  const rgb =
    m7HexToRgb(hex);

  if(!rgb){
    return;
  }

  root.style.setProperty(
    "--m7-accent",
    "#" + String(hex).replace("#","")
  );

  root.style.setProperty(
    "--m7-accent-rgb",
    rgb.r + "," + rgb.g + "," + rgb.b
  );

  root.style.setProperty(
    "--m7-accent-light",
    m7Mix(
      rgb,
      {r:255,g:255,b:255},
      .46
    )
  );

  root.style.setProperty(
    "--m7-accent-dark",
    m7Mix(
      rgb,
      {r:0,g:0,b:0},
      .35
    )
  );

}



/* =========================================================
   FOLLOW / MESSAGE TYPOGRAPHY
   Reads global + Follow typography from Page Design Studio.
========================================================= */

function m7FollowFontStack(mode){

  return (
    {
      system:
        '"Segoe UI",Arial,Helvetica,sans-serif',

      modern:
        '"Trebuchet MS","Segoe UI",Arial,sans-serif',

      elegant:
        'Georgia,"Times New Roman",serif',

      classic:
        '"Times New Roman",Georgia,serif',

      mono:
        '"Courier New",Courier,monospace'
    }[
      String(mode || "")
        .trim()
        .toLowerCase()
    ]
    ||
    ""
  );

}


function m7FollowPercent(value){

  const number =
    Number(value);

  if(!Number.isFinite(number)){
    return 100;
  }

  return Math.max(
    70,
    Math.min(
      150,
      number
    )
  );

}


function m7ApplyFollowTypography(
  options
){

  const globalStyle =
    String(
      options.global_font_style ||
      "current"
    )
      .trim()
      .toLowerCase();


  const moduleStyle =
    String(
      options.follow_font_style ||
      "inherit"
    )
      .trim()
      .toLowerCase();


  const finalStyle =
    moduleStyle === "inherit"
      ? globalStyle
      : moduleStyle;


  const family =
    m7FollowFontStack(
      finalStyle
    );


  const scale =
    (
      m7FollowPercent(
        options.global_font_size
      )
      *
      m7FollowPercent(
        options.follow_font_size
      )
    ) / 10000;


  [
    root,
    ...root.querySelectorAll("*")
  ]
    .forEach(
      function(el){

        if(
          !el ||
          !el.tagName
        ){
          return;
        }


        const tag =
          el.tagName
            .toUpperCase();


        if(
          [
            "STYLE",
            "SCRIPT",
            "SVG",
            "PATH",
            "IMG",
            "VIDEO"
          ].includes(tag)
        ){
          return;
        }


        const directText =
          Array.from(
            el.childNodes || []
          )
            .some(
              function(node){
                return (
                  node.nodeType === 3 &&
                  String(
                    node.textContent || ""
                  ).trim()
                );
              }
            );


        const formText =
          [
            "BUTTON",
            "INPUT",            "TEXTAREA",
            "SELECT",
            "OPTION"
          ].includes(tag);


        if(
          !directText &&
          !formText
        ){
          return;
        }


        if(family){
          el.style.setProperty(
            "font-family",
            family,
            "important"
          );
        }
        else{
          el.style.removeProperty(
            "font-family"
          );
        }


        let base =
          Number(
            el.dataset.m7BaseFontSize
          );


        if(
          !Number.isFinite(base) ||
          base <= 0
        ){

          const current =
            parseFloat(
              getComputedStyle(el)
                .fontSize
            );


          if(
            !Number.isFinite(current) ||
            current <= 0
          ){
            return;
          }


          base =
            current;


          el.dataset.m7BaseFontSize =
            String(base);

        }


        el.style.setProperty(
          "font-size",
          (
            base *
            scale
          )
            .toFixed(2) +
          "px",
          "important"
        );

      }
    );

}


async function loadShopAccent(){

  try{

    const result =
      await client
        .from("shop_profiles")
        .select("directory_options")
        .eq("shop_slug",SHOP_SLUG)
        .maybeSingle();

    if(result.error || !result.data){
      return;
    }

    const options =
      result.data.directory_options &&
      typeof result.data.directory_options === "object"
        ? result.data.directory_options
        : {};

    const universal =
      String(
        options.story_color ||
        options.card_color ||
        ""
      ).trim();


    const useUniversal =
      options.page_use_universal_accent === undefined
        ? true
        : (
            options.page_use_universal_accent === true ||
            String(
              options.page_use_universal_accent
            ).toLowerCase() === "true"
          );


    const accent =
      String(
        useUniversal
          ? universal
          : (
              options.follow_accent_color ||
              universal
            )
      ).trim();


    const preset =
      String(
        options.page_design_preset ||
        "premium"
      )
        .trim()
        .toLowerCase();


    const motionMode =
      String(
        options.page_motion_mode ||
        "preset"
      )
        .trim()
        .toLowerCase();


    const motionOff =
      motionMode === "off" ||
      (
        motionMode === "preset" &&
        (
          preset === "basic" ||
          preset === "minimal"
        )
      );


    root.classList.toggle(
      "m7-basic-static",
      motionOff
    );


    if(accent){
      m7ApplyAccent(accent);
    }


    m7ApplyFollowTypography(
      options
    );

  }catch(error){

    console.warn(
      "[Ma7alak Follow] profile accent:",
      error
    );

  }

}


loadShopAccent();

/* =========================================================
   AUTHENTICATED FOLLOW CLIENT
========================================================= */

async function getFollowClient(){

  const h =
    host();


  try{

    const account =
      h.Ma7alakAccount;


    if(!account){

      return client;

    }


    if(
      typeof
        account.ready ===
      "function"
    ){

      await account.ready();

    }


    /*
       Logged-in Ma7alak viewers
       MUST use the authenticated
       viewer account client.

       This allows owner notifications
       to identify the actual viewer.
    */

    if(
      account.user &&
      account.client &&
      typeof
        account.client.rpc ===
      "function"
    ){

      return account.client;

    }

  }


  catch(error){

    console.warn(
      "[Ma7alak Follow] viewer auth:",
      error
    );

  }


  return client;

}

/* =========================================================
   FOLLOW TIMEOUT UI
========================================================= */

function followRetrySeconds(){

  return Math.max(
    0,
    Math.ceil(
      (
        Number(followBlockedUntil || 0) -
        Date.now()
      ) /
      1000
    )
  );

}


function startFollowTimeout(
  seconds
){

  const duration =
    Math.max(
      1,
      Math.min(
        300,
        Number(seconds) || 60
      )
    );


  followBlockedUntil =
    Date.now() +
    duration * 1000;


  if(followBlockTimer){
    clearInterval(
      followBlockTimer
    );
  }


  followBlockTimer =
    setInterval(
      function(){

        if(
          followRetrySeconds() <= 0
        ){

          clearInterval(
            followBlockTimer
          );

          followBlockTimer =
            null;

          followBlockedUntil =
            0;

        }


        render();

      },
      1000
    );


  render();

}

function render(){


  /* COUNT */

  countText.textContent =
    Math.max(

      Number(followerCount) || 0

      .toLocaleString(

      );

        );
  }


  if(likeCountText){
    likeCountText.textContent =

        Number(likeCount) || 0
      )
        .toLocaleString(
          "en-US"
        );
  }


  if(ownerCountText){
    ownerCountText.textContent =
      followerLabel(
        followerCount
      );
  }


  /* OWNER */

  root
    .classList
    .toggle(
      "is-own-shop",
      isOwnShop
    );


  /* =====================================================
     NOTIFICATION TEXT
  ===================================================== */


  if(
    isOwnShop
  ){

    hintText.textContent =
      "";


    hintText
      .classList
      .remove(
        "visible"
      );

  }


  else if(
    following
  ){

    /*
       FOLLOWING:

       Show ONLY:
       Notifications ON
    */

    hintText.textContent =
      "Notifications ON";


    hintText
      .classList
      .add(
        "visible"
      );

  }


  else{

    /*
       NOT FOLLOWING:

       NOTHING UNDER FOLLOWER COUNT
    */

    hintText.textContent =
      "";


    hintText
      .classList
      .remove(
        "visible"
      );

  }


  /* =====================================================
     BUTTON
  ===================================================== */

  if(
    following
  ){

    button
      .classList
      .add(
        "is-following"
      );


    buttonIcon.textContent =
      "✓";


    buttonText.textContent =
      "Following";


    button.setAttribute(
      "aria-pressed",
      "true"
    );

  }


  else{

    button
      .classList
      .remove(
        "is-following"
      );


    buttonIcon.textContent =
      "+";


    buttonText.textContent =
      "Follow";


    button.setAttribute(
      "aria-pressed",
      "false"
    );

  }


  /* BUSY / ANTI-SPAM TIMEOUT */

  const retrySeconds =
    followRetrySeconds();


  if(
    retrySeconds > 0 &&
    !isOwnShop
  ){

    buttonIcon.textContent =
      "⏳";


    buttonText.textContent =
      "Wait " +
      retrySeconds +
      "s";


    button.setAttribute(
      "aria-label",
      "Follow temporarily paused. Try again in " +
      retrySeconds +
      " seconds."
    );

  }
  else{

    button.removeAttribute(
      "aria-label"
    );

  }


  button
    .classList
    .toggle(
      "is-busy",
      busy
    );


  button.disabled =
    busy ||
    retrySeconds > 0;

}


/* =========================================================
   OWNER CHECK
========================================================= */

async function checkOwnShop(){

  try{

    const h =
      host();


    if(
      h.Ma7alakOwnerAuth
        ?.ready
    ){

      await h
        .Ma7alakOwnerAuth
        .ready();

    }


    const ownedSlug =
      h.Ma7alakOwnerAuth
        ?.owner
        ?.shop_slug;


    isOwnShop =
      !!ownedSlug &&
      String(
        ownedSlug
      ) ===
      SHOP_SLUG;


    return isOwnShop;

  }


  catch(_){

    isOwnShop =
      false;


    return false;

  }

}


/* =========================================================
   LOAD FOLLOW STATE
========================================================= */

async function loadState(){

  const followClient =
    await getFollowClient();


  const result =
    await followClient
      .rpc(
        "get_shop_follow_state",
        {

          p_shop_slug:
            SHOP_SLUG,

          p_visitor_id:
            VISITOR_ID

        }
      );


  if(
    result.error
  ){

    console.warn(
      "[Ma7alak Follow] state:",
      result.error
    );


    busy =
      false;


    render();


    showError(
      "Couldn't load Follow."
    );


    return;

  }


  const row =
    firstRow(
      result.data
    );


  if(row){

    following =
      !!row.following;


    followerCount =
      Math.max(
        0,

        Number(
          row.follower_count
        )

        ||

        0
      );

  }


  busy =
    false;


  render();

}

  const next =
    Math.max(
      0,

        result.data
      )

      ||

      0
    );


  if(

    followerCount
  ){

      next;

  }

/* =========================================================
   LOAD REAL FOLLOWING COUNT

   Uses the SAME installed Follow RPC as the website header.
   This is the total number of shops this visitor identity follows,
   not just a fake 0 / 1 for the shop currently being viewed.
========================================================= */

function normalizeFollowedSlugs(
  data
){

  const rows =
    Array.isArray(data)
      ? data
      : [];


  return Array
    .from(
      new Set(
        rows
          .map(
            function(row){

              if(
                typeof row ===
                "string"
              ){

                return row.trim();

              }


              return String(
                (
                  row &&
                  (
                    row.shop_slug ||                    row.p_shop_slug ||
                    row.slug
                  )
                )
                ||
                ""
              )
                .trim();

            }
          )
          .filter(Boolean)
      )
    );

}

    const result =

/* =========================================================
   FOLLOW / UNFOLLOW
========================================================= */

async function toggleFollow(){

  if(
    busy ||
    isOwnShop ||
    followRetrySeconds() > 0
  ){

    return;

  }


  busy =
    true;


  const oldFollowing =
    following;


  const oldCount =
    followerCount;


  /* =====================================================
     INSTANT UI UPDATE
  ===================================================== */

  following =
    !following;


  if(
    following
  ){

    followerCount =
      oldCount + 1;

  }


  else{

    followerCount =
      Math.max(
        0,
        oldCount - 1
      );

  }


  render();


  broadcastChange();


  /* =====================================================
     DATABASE
  ===================================================== */

  try{

    const followClient =
      await getFollowClient();


    const rpcName =
      following

        ?

        "follow_shop"

        :

        "unfollow_shop";


    const result =
      await followClient
        .rpc(
          rpcName,
          {

            p_shop_slug:
              SHOP_SLUG,

            p_visitor_id:
              VISITOR_ID

          }
        );


    if(
      result.error
    ){

      throw result.error;

    }


    const row =
      firstRow(
        result.data
      );


    /*
       Use exact values
       returned by Supabase.
    */

    if(row){


      if(
        typeof
          row.following ===
        "boolean"
      ){

        following =
          row.following;

      }


      if(
        row.follower_count !==
          undefined &&
        row.follower_count !==
          null
      ){

        followerCount =
          Math.max(
            0,

            Number(
              row.follower_count
            )

            ||

            0
          );

      }

    }


    busy =
      false;


    render();


    broadcastChange();


    /*
       Confirm exact DB count.
    */

    setTimeout(
      loadCount,
      100
    );


    setTimeout(
      loadFollowingCount,
      120
    );

  }


  catch(error){

    /*
       RESTORE OLD STATE
       IF RPC FAILED
    */

    following =
      oldFollowing;


    followerCount =
      oldCount;


    busy =
      false;


    render();


    broadcastChange();


    console.warn(
      "[Ma7alak Follow]",
      error
    );


    const errorMessage =
      String(
        error?.message ||
        ""
      );


    const timeoutMatch =
      errorMessage.match(
        /Follow changes paused\. Try again in\s+(\d+)\s+seconds/i
      );


    if(timeoutMatch){

      const seconds =
        Math.max(
          1,
          Number(timeoutMatch[1]) || 60
        );


      startFollowTimeout(
        seconds
      );


      showError(
        "Too many Follow changes. Try again in " +
        seconds +
        " seconds."
      );

    }
    else{

      showError(
        "Couldn't update Follow. Try again."
      );

    }

  }

}

/* =========================================================
   EXTERNAL FOLLOW CHANGE

   This is what keeps this box synced
   when user unfollows Zee from the
   HEADER FOLLOWING PANEL.
========================================================= */

function externalFollowChange(
  event
){

  const detail =
    event?.detail ||
    {};


  const eventSlug =
    String(

      detail.shopSlug

      ||

      detail.shop_slug

      ||

      ""

    )
    .trim();


  if(
    !eventSlug ||
    eventSlug !==
      SHOP_SLUG
  ){

    return;

  }


  /* =====================================================
     DUPLICATE EVENT PROTECTION
  ===================================================== */

  const now =
    Date.now();


  if(
    now -
    lastExternalSync <
    200
  ){

    return;

  }


  lastExternalSync =
    now;


  const oldFollowing =
    following;


  /* =====================================================
     FOLLOW STATE
  ===================================================== */

  if(
    typeof
      detail.following ===
    "boolean"
  ){

    following =
      detail.following;

  }


  /* =====================================================
     EXACT COUNT IF PROVIDED
  ===================================================== */

  const providedCount =

    detail.followerCount

    ??

    detail.follower_count;


  if(
    providedCount !==
      undefined &&

    providedCount !==
      null &&

    !Number.isNaN(
      Number(
        providedCount
      )
    )
  ){

    followerCount =
      Math.max(
        0,

        Number(
          providedCount
        )
      );

  }


  /* =====================================================
     NO COUNT PROVIDED
     UPDATE LOCALLY
  ===================================================== */

  else if(
    oldFollowing &&
    !following
  ){

    followerCount =
      Math.max(
        0,
        followerCount - 1
      );

  }


  else if(
    !oldFollowing &&
    following
  ){

    followerCount =
      followerCount + 1;

  }


  busy =
    false;


  /*
     IMMEDIATE RESULT:

     ✓ Following becomes + Follow
     ✓ count updates
     ✓ Notifications ON disappears
  */

  render();


  /* confirm DB */

  setTimeout(
    loadCount,
    100
  );


  setTimeout(
    loadFollowingCount,
    120
  );


  setTimeout(
    loadState,
    300
  );

}

/* =========================================================
   REALTIME FOLLOWER COUNT
========================================================= */

function startRealtime(){

  realtimeChannel =
    client

      .channel(

        "ma7alak-follow-"

        +

        SHOP_SLUG.replace(
          /[^A-Za-z0-9_-]/g,
          "_"
        )

        +

        "-"

        +

        Math.random()
          .toString(36)
          .slice(2,8)

      )


      .on(

        "postgres_changes",

        {

          event:"*",

          schema:"public",

          table:
            "shop_follows",

          filter:
            "shop_slug=eq."
            +
            SHOP_SLUG

        },


        function(){

          loadCount();

          loadFollowingCount();

        }

      )


      .subscribe();


  /*
    The profile owner's Following total can change when that owner
    follows/unfollows ANY OTHER shop, not only this shop.
    Listen to follow-table changes and refresh only the owner count.
  */
  if(
    !ownerFollowingChannel &&
    typeof client.channel ===
      "function"
  ){

    ownerFollowingChannel =
      client
        .channel(
          "ma7alak-owner-following-" +
          SHOP_SLUG +
          "-" +
          Math.random()
            .toString(36)
            .slice(2,8)
        )
        .on(
          "postgres_changes",
          {
            event:"*",
            schema:"public",
            table:"shop_follows"
          },
          function(){
            setTimeout(
              loadFollowingCount,
              80
            );
          }
        )
        .subscribe();

  }

}

  fallbackTimer =

        if(
          document.visibilityState ===

          loadCount();

          loadFollowingCount();

    .addEventListener(

      function(){

        if(
          document.visibilityState ===
          "visible"
        ){

          checkOwnShop()
            .then(
              loadState
            )
            .then(
              loadFollowingCount
            );

        }

      }
    );

}
