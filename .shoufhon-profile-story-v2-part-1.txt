  function ma7alakStoryHexToRgb(hex){

    const clean =
      String(hex || "")
        .trim()
        .replace("#","");

    if(!/^[0-9a-f]{6}$/i.test(clean)){
      return null;
    }

    return {
      r:parseInt(clean.slice(0,2),16),
      g:parseInt(clean.slice(2,4),16),
      b:parseInt(clean.slice(4,6),16)
    };
  }


  function ma7alakStoryMixHex(
    rgb,
    target,
    amount
  ){

    function clamp(value){
      return Math.max(
        0,
        Math.min(
          255,
          Math.round(value)
        )
      );
    }

    return (
      "#" +
      [
        clamp(rgb.r + (target.r-rgb.r)*amount),
        clamp(rgb.g + (target.g-rgb.g)*amount),
        clamp(rgb.b + (target.b-rgb.b)*amount)
      ]
        .map(
          function(value){
            return value
              .toString(16)
              .padStart(2,"0");
          }
        )
        .join("")
    );
  }


  function ma7alakStoryRgba(
    rgb,
    alpha
  ){

    return (
      "rgba(" +
      rgb.r + "," +
      rgb.g + "," +
      rgb.b + "," +
      alpha +
      ")"
    );
  }

    const own =
      String(custom || "").trim();

    if(
      kind === "custom" &&
      own
    ){
      return own.slice(0,5);
    }

    return (
      {
        diamond:"◆",
        star:"★",
        sparkle:"✦",
        dot:"•",
        none:""
      }[kind] ||
      "◆"
    );
  }


  function m7dsLineBackground(
    style,
    color,
    rgba
  ){

    if(style === "solid"){
      return color;
    }

    if(style === "double"){
      return (
        "linear-gradient(to bottom," +
        color +
        " 0 1px,transparent 1px 2px," +
        color +
        " 2px 3px)"
      );
    }

    if(style === "dotted"){
      return (
        "radial-gradient(circle," +
        color +
        " 0 1.2px,transparent 1.4px) 0 50% / 7px 3px repeat-x"
      );
    }

    return (
      "linear-gradient(90deg,transparent 0%," +
      rgba +
      " 24%," +
      color +
      " 100%)"
    );
  }



  /* =========================================================
     PAGE TYPOGRAPHY
     Global + per-module font style and proportional text size.
     No external font downloads are used.
  ========================================================= */

  function m7dsFontStack(
    mode
  ){

    return (
      {
        system:
          '"Segoe UI",Arial,Helvetica,sans-serif',

        modern:
          '"Trebuchet MS","Segoe UI",Arial,sans-serif',

        elegant:
          'Georgia,"Times New Roman",serif',

        classic:
          '"Times New Roman",Georgia,serif',

        mono:
          '"Courier New",Courier,monospace'
      }[
        String(mode || "")
          .trim()
          .toLowerCase()
      ]
      ||
      ""
    );
  }


  function m7dsFontPercent(
    value
  ){

    const number =
      Number(value);

    if(
      !Number.isFinite(number)
    ){
      return 100;
    }

    return Math.max(
      70,
      Math.min(
        150,
        number
      )
    );
  }


  function m7dsApplyTypography(
    root,
    options,
    moduleKey
  ){

    if(!root){
      return;
    }


    const globalStyle =
      String(
        options.global_font_style ||
        "current"
      )
        .trim()
        .toLowerCase();


    const moduleStyleRaw =
      String(
        options[
          moduleKey +
          "_font_style"
        ] ||
        "inherit"
      )
        .trim()
        .toLowerCase();


    const finalStyle =
      moduleStyleRaw === "inherit"
        ? globalStyle
        : moduleStyleRaw;


    const family =
      m7dsFontStack(
        finalStyle
      );


    const globalScale =
      m7dsFontPercent(
        options.global_font_size
      );


    const moduleScale =
      m7dsFontPercent(
        options[
          moduleKey +
          "_font_size"
        ]
      );


    const finalScale =
      (
        globalScale *
        moduleScale
      ) / 10000;


    const elements =
      [
        root,
        ...root.querySelectorAll("*")
      ];


    elements.forEach(
      function(el){

        if(
          !el ||
          !el.tagName
        ){
          return;
        }


        const tag =
          el.tagName
            .toUpperCase();


        if(
          [
            "STYLE",
            "SCRIPT",
            "SVG",
            "PATH",
            "IMG",
            "VIDEO"
          ].includes(tag)
        ){
          return;
        }


        const directText =
          Array.from(
            el.childNodes || []
          )
            .some(
              function(node){
                return (
                  node.nodeType === 3 &&
                  String(
                    node.textContent || ""
                  ).trim()
                );
              }
            );


        const formText =
          [
            "BUTTON",
            "INPUT",
            "TEXTAREA",
            "SELECT",
            "OPTION"
          ].includes(tag);


        if(
          !directText &&
          !formText
        ){
          return;
        }


        if(
          family
        ){
          el.style.setProperty(
            "font-family",
            family,
            "important"
          );
        }
        else{
          el.style.removeProperty(
            "font-family"
          );
        }


        let base =
          Number(
            el.dataset
              .m7BaseFontSize
          );


        if(
          !Number.isFinite(base) ||
          base <= 0
        ){

          const current =
            parseFloat(
              getComputedStyle(el)
                .fontSize
            );


          if(
            !Number.isFinite(current) ||
            current <= 0
          ){
            return;
          }


          base =
            current;


          el.dataset
            .m7BaseFontSize =
              String(base);

        }


        el.style.setProperty(
          "font-size",
          (
            base *
            finalScale
          )
            .toFixed(2) +
          "px",
          "important"
        );

      }
    );

  }

  function storyURL(path){

  /* =========================================================
     STORY MEDIA WARM-UP — INTENT BASED
     ---------------------------------------------------------
     Do NOT download Story media just because the shop page loaded.

     Once a visitor actually opens Stories, only the NEXT Story is
     warmed. Videos warm metadata only; the active Story itself loads
     normally when shown. This keeps swipe transitions responsive while
     avoiding unnecessary Storage egress for visitors who never open
     Stories.
  ========================================================= */

  const ma7alakStoryWarmCache =
    new Map();

  function warmStoryMedia(story){

    if(
      !story ||
      !story.storage_path
    ){
      return;
    }

    const url =
      storyURL(
        story.storage_path
      );

    if(
      ma7alakStoryWarmCache.has(
        url
      )
    ){
      return;
    }

    const type =
      mediaType(
        story
      );

    if(
      type === "video"
    ){
      const video =
        document.createElement(
          "video"
        );

      video.preload =
        "metadata";

      video.muted =
        true;

      video.playsInline =
        true;

      video.setAttribute(
        "playsinline",
        ""
      );

      video.src =
        url;

      ma7alakStoryWarmCache.set(
        url,
        video
      );

      try{
        video.load();
      }
      catch(error){}
    }
    else{
      const image =
        new Image();

      image.decoding =
        "async";

      image.src =
        url;

      ma7alakStoryWarmCache.set(
        url,
        image
      );

      if(
        typeof image.decode ===
        "function"
      ){
        image.decode().catch(
          function(){}
        );
      }
    }

  }

  function pruneStoryWarmCache(){

    const activeUrls =
      new Set(
        (Array.isArray(stories) ? stories : [])
          .filter(function(story){
            return story && story.storage_path;
          })
          .map(function(story){
            return storyURL(
              story.storage_path
            );
          })
      );


    ma7alakStoryWarmCache.forEach(
      function(media,url){

        if(
          activeUrls.has(url)
        ){
          return;
        }


        try{

          if(
            media &&
            media.tagName === "VIDEO"
          ){
            media.pause();
            media.removeAttribute(
              "src"
            );
            media.load();
          }
          else if(media){
            media.src = "";
          }

        }
        catch(error){}


        ma7alakStoryWarmCache.delete(
          url
        );

      }
    );

  }


  function warmNextStoryMedia(){

    const screen =
      document.getElementById(
        "ma7alak-full-story"
      );


    if(
      document.hidden ||
      !screen ||
      !screen.classList.contains(
        "active"
      ) ||
      !Array.isArray(stories) ||
      stories.length < 2
    ){
      return;
    }


    const nextIndex =
      (
        currentStory + 1
      ) %
      stories.length;


    warmStoryMedia(
      stories[nextIndex]
    );

  }


  function scheduleStoryExpiryRefresh(){

    if(
      ma7alakStoryExpiryTimer
    ){
      clearTimeout(
        ma7alakStoryExpiryTimer
      );

      ma7alakStoryExpiryTimer =
        null;
    }


    let nextExpiry =
      Infinity;

    const now =
      Date.now();


    (Array.isArray(stories) ? stories : [])
      .forEach(
        function(story){

          const expires =
            new Date(
              story &&
              story.expires_at ||
              ""
            ).getTime();


          if(
            Number.isFinite(expires) &&
            expires > now &&
            expires < nextExpiry
          ){
            nextExpiry =
              expires;
          }

        }
      );


    if(
      !Number.isFinite(
        nextExpiry
      )
    ){
      return;
    }


    const delay =
      Math.max(
        1000,
        Math.min(
          2147483000,
          nextExpiry -
          now +
          1200
        )
      );


    ma7alakStoryExpiryTimer =
      setTimeout(
        function(){

          ma7alakStoryExpiryTimer =
            null;


          if(
            document.visibilityState ===
            "visible"
          ){
            loadStories();
          }

        },
        delay
      );

  }

  /* =========================================================
     MEDIA TYPE
  ========================================================= */

  function mediaType(story){

    if(
      story.media_type ===
      "video"
    ){

      return "video";

    }


    if(
      story.media_type ===
      "image"
    ){

      return "image";

    }


    const path =
      String(
        story.storage_path ||
        ""
      ).toLowerCase();


    if(
      path.endsWith(".mp4") ||
      path.endsWith(".webm") ||
      path.endsWith(".mov") ||
      path.endsWith(".m4v")
    ){

      return "video";

    }


    return "image";

  }


  /* =========================================================
     TIME AGO
  ========================================================= */

  function storyTimeAgo(createdAt){

    const created =
      new Date(createdAt).getTime();

    if(!Number.isFinite(created)){
      return "Just now";
    }

    const now = Date.now();

    let seconds =
      Math.floor(
        (now - created) /
        1000
      );

    if(seconds < 0){
      seconds = 0;
    }

    if(seconds < 10){
      return "Just now";
    }

    if(seconds < 60){
      return seconds + (seconds === 1 ? " second ago" : " seconds ago");
    }

    const minutes =
      Math.floor(seconds / 60);

    if(minutes < 60){
      return minutes + (minutes === 1 ? " minute ago" : " minutes ago");
    }

    const hours =
      Math.floor(minutes / 60);

    if(hours < 24){
      return hours + (hours === 1 ? " hour ago" : " hours ago");
    }

    const days =
      Math.floor(hours / 24);

    if(days === 1){
      return "Yesterday";
    }

    if(days < 7){
      return days + " days ago";
    }

    const weeks =
      Math.floor(days / 7);

    if(weeks === 1){
      return "1 week ago";
    }

    return weeks + " weeks ago";

  }


  /* =========================================================
     UPDATE STORY TIME
  ========================================================= */

  function updateStoryTime(){

    const timeElement =
      document.getElementById(
        "ma7alak-story-time"
      );

    const nameElement =
      document.getElementById(
        "ma7alak-story-shop-name"
      );


    const statusElement =
      document.getElementById(
        "ma7alak-story-status-text"
      );


    if(
      !stories.length
    ){

      return;

    }


    const story =
      stories[
        currentStory
      ];


    if(!story){

      return;

    }


    if(timeElement){

      timeElement.textContent =
        storyTimeAgo(
          story.created_at
        );

    }


    if(nameElement){

      const fallbackShopName =
        String(story.shop_slug || SHOP_SLUG)
          .split("-")
          .filter(Boolean)
          .map(function(part){
            return part.charAt(0).toUpperCase() + part.slice(1);
          })
          .join(" ");

      const shopName =
        String(
          SHOP_IDENTITY.name ||
          fallbackShopName ||
          "Ma7alak"
        ).trim();

      nameElement.textContent =
        shopName || "Ma7alak";

      /*
        Every Story already has its own unique Supabase
        story.id. Keep that ID attached to the displayed
        shop name so the name always belongs to the exact
        Story currently open.
      */
      nameElement.setAttribute(
        "data-story-id",
        String(story.id)
      );

    }


    if(statusElement){

      const statusText =
        String(
          story.status_text || ""
        ).trim();

      statusElement.textContent =
        statusText;

      statusElement.classList.toggle(
        "visible",
        Boolean(statusText)
      );

    }

  }


  /* =========================================================

  ========================================================= */

  function startNewStoryAnimation(){

    if(!storyWrapper){
      return;
    }

    storyWrapper.classList.remove(
      "story-has-new"
    );

    /* Restart CSS animation reliably on mobile browsers. */
    void storyWrapper.offsetWidth;

    storyWrapper.classList.add(
      "story-has-new"
    );

  }


  /* =========================================================
     STOP NEW STORY ANIMATION
  ========================================================= */

  function stopNewStoryAnimation(){

    if(
      storyWrapper
    ){

      storyWrapper.classList.remove(
        "story-has-new"
      );

    }


    if(newRingAnimation){
      try{newRingAnimation.cancel();}catch(error){}
      newRingAnimation=null;
    }

  }


  /* =========================================================
     CHECK NEW STORY
  ========================================================= */

  function updateNewStoryAnimation(){

    if(!storyWrapper){

      return;

    }


    if(
      !stories.length
    ){

      stopNewStoryAnimation();

      return;

    }


    const newestStory =
      stories[
        stories.length - 1
      ];


    let lastSeen =
      null;


    try{

      lastSeen =
        localStorage.getItem(
          STORY_SEEN_KEY
        );

    }

    catch(error){

      lastSeen =
        null;

    }


    if(!lastSeen){

      startNewStoryAnimation();

      return;

    }


    const newestTime =
      new Date(
        newestStory.created_at
      ).getTime();


    const lastSeenTime =
      new Date(
        lastSeen
      ).getTime();


    if(
      newestTime >
      lastSeenTime
    ){

      startNewStoryAnimation();

    }

    else{

      stopNewStoryAnimation();

    }

  }


  /* =========================================================
     MARK STORIES AS SEEN
  ========================================================= */

  function getFirstUnseenStoryIndex(){

    if(!stories.length){
      return 0;
    }

    let lastSeen = null;

    try{
      lastSeen =
        localStorage.getItem(
          STORY_SEEN_KEY
        );
    }
    catch(error){}

    /*
       First visit: start from the oldest Story.
    */
    if(!lastSeen){
      return 0;
    }

    const lastSeenTime =
      new Date(lastSeen).getTime();

    if(!Number.isFinite(lastSeenTime)){
      return 0;
    }

    for(
      let index = 0;
      index < stories.length;
      index++
    ){

      const storyTime =
        new Date(
          stories[index].created_at
        ).getTime();

      if(
        Number.isFinite(storyTime) &&
        storyTime > lastSeenTime
      ){
        return index;
      }

    }

    /*
       Everything has already been seen.
       Start from the first Story again.
    */
    return 0;

  }


  /* =========================================================
     MARK CURRENT STORY AS SEEN
     ---------------------------------------------------------
     Only the Story that is actually opened is marked seen.
     If more new Stories remain, the new-story ring stays active.
  ========================================================= */

  function markCurrentStoryAsSeen(){

    if(!stories.length){
      return;
    }

    const story =
      stories[currentStory];

    if(
      !story ||
      !story.created_at
    ){
      return;
    }

    let lastSeen = null;

    try{
      lastSeen =
        localStorage.getItem(
          STORY_SEEN_KEY
        );
    }
    catch(error){}

    const storyTime =
      new Date(
        story.created_at
      ).getTime();

    const lastSeenTime =
      lastSeen
        ? new Date(lastSeen).getTime()
        : 0;

    if(
      Number.isFinite(storyTime) &&
      (
        !Number.isFinite(lastSeenTime) ||
        storyTime > lastSeenTime
      )
    ){

      try{
        localStorage.setItem(
          STORY_SEEN_KEY,
          story.created_at
        );
      }
      catch(error){}

    }

    updateNewStoryAnimation();

  }

  function applyParentOwnerState(data){

    if(
      !data ||
      data.type !== "MA7ALAK_OWNER_STATE"
    ){
      return false;
    }

    const incomingSlug =
      String(data.shopSlug || "")
        .trim()
        .toLowerCase();

    const currentSlug =
      String(SHOP_SLUG || "")
        .trim()
        .toLowerCase();

    ma7alakParentOwnerStateReceived = true;
    ma7alakParentIsOwner = !!data.isOwner;
    ma7alakParentOwnerSlug = incomingSlug;

    setShopOwnerState(
      ma7alakParentIsOwner &&
      incomingSlug === currentSlug
    );

    return true;

  }

  function ownerPageTarget(){

    try{
      if(
        window.top &&
        window.top !== window
      ){
        return window.top;
      }
    }
    catch(error){}

    try{
      if(
        window.parent &&
        window.parent !== window
      ){
        return window.parent;
      }
    }
    catch(error){}

    return null;

  }
